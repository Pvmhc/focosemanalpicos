const NOME_SEPARADOR = 'gonçalo'; const NOME_CONFIG = 'config'; const CABECALHO = ['id', 'cliente', 'numero', 'comercial', 'dia', 'semana', 'estadosJSON', 'historicoJSON', 'colocouJSON']; const CABECALHO_CONFIG = ['chave', 'valor']; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(NOME_SEPARADOR); if (!sheet) { sheet = ss.insertSheet(NOME_SEPARADOR); } if (sheet.getLastRow() === 0) { sheet.appendRow(CABECALHO); } // Migração automática: se a folha já existia sem a coluna colocouJSON, acrescenta-a const cabecalhoAtual = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getValues()[0].map(h => String(h).trim().toLowerCase()); if (cabecalhoAtual.indexOf('colocoujson') === -1) { sheet.getRange(1, sheet.getLastColumn() + 1).setValue('colocouJSON'); } return sheet; } function getConfigSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(NOME_CONFIG); if (!sheet) { sheet = ss.insertSheet(NOME_CONFIG); } if (sheet.getLastRow() === 0) { sheet.appendRow(CABECALHO_CONFIG); } return sheet; } function lerConfig_(chave) { const sheet = getConfigSheet_(); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (String(dados[i][0]) === chave) { try { return JSON.parse(dados[i][1] || 'null'); } catch (err) { return null; } } } return null; } function gravarConfig_(chave, valor) { const sheet = getConfigSheet_(); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (String(dados[i][0]) === chave) { sheet.getRange(i + 1, 2).setValue(JSON.stringify(valor)); return; } } sheet.appendRow([chave, JSON.stringify(valor)]); } function saidaJSON_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** Devolve {idx, ultimaLinha} com as colunas do cabeçalho em minúsculas, sem ler a folha toda */ function obterCabecalhoIdx_(sheet) { const ultimaLinha = sheet.getLastRow(); const ultimaColuna = Math.max(sheet.getLastColumn(), 1); const cabecalho = sheet.getRange(1, 1, 1, ultimaColuna).getValues()[0].map(h => String(h).trim().toLowerCase()); const idx = {}; cabecalho.forEach((h, i) => { idx[h] = i + 1; }); // 1-based (coluna de sheet) return { idx, ultimaLinha }; } /** GET — devolve todos os registos + configuração (marcaFocoPorSemana) num único pedido, * ou só a configuração se ?tipo=config (mantido para compatibilidade/outros usos) */ function doGet(e) { try { if (e && e.parameter && e.parameter.tipo === 'config') { const marcaFocoPorSemana = lerConfig_('marcaFocoPorSemana'); const ordemMarcas = lerConfig_('ordemMarcas'); return saidaJSON_({ ok: true, marcaFocoPorSemana: marcaFocoPorSemana || {}, ordemMarcas: ordemMarcas || [] }); } const sheet = getSheet_(); const dados = sheet.getDataRange().getValues(); const marcaFocoPorSemana = lerConfig_('marcaFocoPorSemana') || {}; const ordemMarcas = lerConfig_('ordemMarcas') || []; if (!dados || dados.length <= 1) { return saidaJSON_({ registos: [], marcaFocoPorSemana: marcaFocoPorSemana, ordemMarcas: ordemMarcas }); } const cabecalho = dados.shift().map(h => String(h).trim().toLowerCase()); const idx = {}; cabecalho.forEach((h, i) => { idx[h] = i; }); const linhas = dados .filter(r => idx['id'] !== undefined && r[idx['id']] !== '' && r[idx['id']] !== null && r[idx['id']] !== undefined) .map(r => { let estados = {}; let historico = {}; let colocou = {}; try { estados = JSON.parse(r[idx['estadosjson']] || '{}'); } catch (err) { estados = {}; } try { historico = JSON.parse(r[idx['historicojson']] || '{}'); } catch (err) { historico = {}; } try { colocou = idx['colocoujson'] !== undefined ? JSON.parse(r[idx['colocoujson']] || '{}') : {}; } catch (err) { colocou = {}; } return { id: r[idx['id']], cliente: r[idx['cliente']], numero: idx['numero'] !== undefined ? r[idx['numero']] : '', comercial: r[idx['comercial']], dia: r[idx['dia']], semana: String(r[idx['semana']]), estados: estados, historico: historico, colocou: colocou }; }); return saidaJSON_({ registos: linhas, marcaFocoPorSemana: marcaFocoPorSemana, ordemMarcas: ordemMarcas }); } catch (err) { return saidaJSON_({ ok: false, erro: 'Erro no servidor (GET): ' + err.message }); } } /** POST — recebe ações do HTML */ function doPost(e) { const lock = LockService.getScriptLock(); let temLock = false; try { temLock = lock.tryLock(30000); if (!temLock) { return saidaJSON_({ ok: false, erro: 'Servidor ocupado. Tenta novamente.' }); } let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return saidaJSON_({ ok: false, erro: 'Corpo do pedido inválido' }); } if (body.action === 'limpar_e_importar') { return acaoLimparEImportar_(body.rows || []); } if (body.action === 'atualizar_estado') { return acaoAtualizarEstado_(body.id, body.marca, body.novoEstado, body.colocou); } if (body.action === 'atualizar_colocou') { return acaoAtualizarColocou_(body.id, body.marca, body.colocado); } if (body.action === 'limpar_foco_semanal') { return acaoLimparFocoSemanal_(body.semana, body.marcaFoco); } if (body.action === 'guardar_config') { gravarConfig_(body.chave, body.valor); return saidaJSON_({ ok: true }); } return saidaJSON_({ ok: false, erro: 'Ação desconhecida: ' + body.action }); } catch (err) { return saidaJSON_({ ok: false, erro: 'Erro no servidor (POST): ' + err.message }); } finally { if (temLock) lock.releaseLock(); } } /** * Limpa TODAS as marcações (FOCO / FOCO SEMANAL / OK / NÃO OK / NEGOCIADO) de * TODOS os registos — todos os dias, todas as semanas, todos os GC/comerciais. * Não mexe em "colocouJSON" (histórico de produto efetivamente colocado). * Pensada para correr automaticamente via gatilho (ver criarGatilhoLimpezaSemanal). */ function apagarTodosOsFocosSemanal() { const lock = LockService.getScriptLock(); let temLock = false; try { temLock = lock.tryLock(30000); if (!temLock) { Logger.log('Limpeza semanal automática: não foi possível obter lock, tenta novamente mais tarde.'); return; } const sheet = getSheet_(); const ultimaLinha = sheet.getLastRow(); if (ultimaLinha < 2) { Logger.log('Limpeza semanal automática: sem registos para limpar.'); return; } const numLinhas = ultimaLinha - 1; const { idx } = obterCabecalhoIdx_(sheet); const idxEstados = idx['estadosjson']; if (!idxEstados) { Logger.log('Limpeza semanal automática: coluna estadosJSON não encontrada.'); return; } const vazio = Array.from({ length: numLinhas }, () => ['{}']); sheet.getRange(2, idxEstados, numLinhas, 1).setValues(vazio); Logger.log('Limpeza semanal automática concluída: ' + numLinhas + ' registos limpos.'); } catch (err) { Logger.log('Erro na limpeza semanal automática: ' + err.message); } finally { if (temLock) lock.releaseLock(); } } /** * Corre isto UMA VEZ manualmente no editor do Apps Script (seleciona esta * função na lista e clica em "Executar") para instalar o gatilho semanal. * Depois disso já não precisas de fazer nada — corre sozinho todas as * segundas-feiras por volta das 08:00, mesmo sem ninguém ter o site aberto. */ function criarGatilhoLimpezaSemanal() { // Remove gatilhos antigos desta função, para nunca ficar duplicado ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'apagarTodosOsFocosSemanal') { ScriptApp.deleteTrigger(t); } }); ScriptApp.newTrigger('apagarTodosOsFocosSemanal') .timeBased() .onWeekDay(ScriptApp.WeekDay.MONDAY) .atHour(8) .create(); Logger.log('Gatilho criado: todas as segundas-feiras às 08:00 (fuso horário do projeto).'); } function acaoLimparEImportar_(rows) { const sheet = getSheet_(); const dadosAntigos = sheet.getDataRange().getValues(); const colocouPorId = {}; if (dadosAntigos.length > 1) { const cabecalhoAntigo = dadosAntigos[0].map(h => String(h).trim().toLowerCase()); const idxIdAntigo = cabecalhoAntigo.indexOf('id'); const idxColocouAntigo = cabecalhoAntigo.indexOf('colocoujson'); if (idxIdAntigo !== -1 && idxColocouAntigo !== -1) { for (let i = 1; i < dadosAntigos.length; i++) { const idAntigo = dadosAntigos[i][idxIdAntigo]; if (idAntigo !== '' && idAntigo !== null && idAntigo !== undefined) { colocouPorId[String(idAntigo)] = dadosAntigos[i][idxColocouAntigo] || '{}'; } } } } sheet.clearContents(); sheet.appendRow(CABECALHO); if (rows.length > 0) { const valores = rows.map(r => { let colocouStr; if (r.colocou !== undefined && r.colocou !== null) { colocouStr = JSON.stringify(r.colocou); } else { colocouStr = colocouPorId[String(r.id)] || '{}'; } return [ r.id, r.cliente, r.numero || '', r.comercial, r.dia, r.semana, JSON.stringify(r.estados || {}), JSON.stringify(r.historico || {}), colocouStr ]; }); sheet.getRange(2, 1, valores.length, CABECALHO.length).setValues(valores); } return saidaJSON_({ ok: true, importados: rows.length }); } /** * Atualiza o estado de UMA marca de UM registo. * Otimizado: em vez de ler a folha inteira (getDataRange), lê apenas a * coluna "id" para localizar a linha, e depois só toca nas 1-2 células * necessárias. Reduz drasticamente o tempo de escrita quando há muitas * linhas/colunas e muitos utilizadores em simultâneo. */ function acaoAtualizarEstado_(id, marca, novoEstado, colocouVal) { const sheet = getSheet_(); const { idx, ultimaLinha } = obterCabecalhoIdx_(sheet); if (ultimaLinha < 2) return saidaJSON_({ ok: false, erro: 'Sem dados' }); const idxId = idx['id']; const idxEstados = idx['estadosjson']; const idxColocou = idx['colocoujson']; const numLinhas = ultimaLinha - 1; const idsColuna = sheet.getRange(2, idxId, numLinhas, 1).getValues(); let linha = -1; for (let i = 0; i < idsColuna.length; i++) { if (String(idsColuna[i][0]) === String(id)) { linha = i + 2; break; } } if (linha === -1) return saidaJSON_({ ok: false, erro: 'ID não encontrado: ' + id }); const celEstados = sheet.getRange(linha, idxEstados); let estados = {}; try { estados = JSON.parse(celEstados.getValue() || '{}'); } catch (err) { estados = {}; } estados[marca] = novoEstado; celEstados.setValue(JSON.stringify(estados)); if (colocouVal !== undefined && colocouVal !== null && idxColocou) { const celColocou = sheet.getRange(linha, idxColocou); let colocou = {}; try { colocou = JSON.parse(celColocou.getValue() || '{}'); } catch (err) { colocou = {}; } colocou[marca] = colocouVal; celColocou.setValue(JSON.stringify(colocou)); } return saidaJSON_({ ok: true }); } function acaoAtualizarColocou_(id, marca, colocado) { const sheet = getSheet_(); const { idx, ultimaLinha } = obterCabecalhoIdx_(sheet); if (ultimaLinha < 2) return saidaJSON_({ ok: false, erro: 'Sem dados' }); const idxId = idx['id']; const idxColocou = idx['colocoujson']; const numLinhas = ultimaLinha - 1; const idsColuna = sheet.getRange(2, idxId, numLinhas, 1).getValues(); let linha = -1; for (let i = 0; i < idsColuna.length; i++) { if (String(idsColuna[i][0]) === String(id)) { linha = i + 2; break; } } if (linha === -1) return saidaJSON_({ ok: false, erro: 'ID não encontrado: ' + id }); const celColocou = sheet.getRange(linha, idxColocou); let colocou = {}; try { colocou = JSON.parse(celColocou.getValue() || '{}'); } catch (err) { colocou = {}; } colocou[marca] = colocado; celColocou.setValue(JSON.stringify(colocou)); return saidaJSON_({ ok: true }); } /** * Limpa, numa ÚNICA leitura/escrita em lote, a marca "FOCOSEMANAL" de * qualquer marca (numa dada semana) que já não seja a marca foco atual. * Substitui o antigo comportamento de disparar 1 pedido POST por registo * alterado (que sobrecarregava o servidor com muitos utilizadores). */ function acaoLimparFocoSemanal_(semana, marcaFoco) { const sheet = getSheet_(); const { idx, ultimaLinha } = obterCabecalhoIdx_(sheet); if (ultimaLinha < 2) return saidaJSON_({ ok: true, alterados: 0 }); const idxSemana = idx['semana']; const idxEstados = idx['estadosjson']; const numLinhas = ultimaLinha - 1; const semanaVals = sheet.getRange(2, idxSemana, numLinhas, 1).getValues(); const estadosVals = sheet.getRange(2, idxEstados, numLinhas, 1).getValues(); const marcaFocoNorm = String(marcaFoco || '').trim().toLowerCase(); let alterados = 0; const novosEstados = estadosVals.map((linhaVal, i) => { if (String(semanaVals[i][0]) !== String(semana)) return linhaVal; let estados = {}; try { estados = JSON.parse(linhaVal[0] || '{}'); } catch (err) { estados = {}; } let mudou = false; Object.keys(estados).forEach(m => { if (estados[m] === 'FOCOSEMANAL' && m.trim().toLowerCase() !== marcaFocoNorm) { estados[m] = ''; mudou = true; } }); if (mudou) { alterados++; return [JSON.stringify(estados)]; } return linhaVal; }); sheet.getRange(2, idxEstados, numLinhas, 1).setValues(novosEstados); return saidaJSON_({ ok: true, alterados: alterados }); }
